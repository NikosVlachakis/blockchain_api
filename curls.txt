# ---------------------------------------------------------------------------
# Direct /client flow as kkech@athenarc.gr (credentials read from .env)
# Run from the repo root; the session cookie is kept in /tmp/fabric-cookies.txt
# LogQuery args (all sent as strings): datasetID, queryBody, epsilonUsed (float)
#   set -a; source .env; set +a
# ---------------------------------------------------------------------------

# 1. Initialize the Fabric gateway for this session (stores the fabric-session cookie)
curl -i -c /tmp/fabric-cookies.txt http://localhost:3000/client/ -H "Content-Type: application/json" -d '{"enrollmentId": "kkech@athenarc.gr", "secret": "BcSecret_1759349573873160992"}'

# 1b. Fetch the Fabric client ID. The chaincode keys budgets and logs by this value -
#     base64 of "x509::<subject>::<issuer>" - NOT by the e-mail address.
#     Note the "PrivacyBudgetContract:" prefix: QueryContract is the default contract,
#     so every PrivacyBudgetContract function must be qualified or you get
#     "Function <name> not found in contract QueryContract".
USER_ID=$(curl -s -b /tmp/fabric-cookies.txt "http://localhost:3000/client/query?channelid=dt4h&chaincodeid=dt4hCC&function=PrivacyBudgetContract:WhoAmI" | sed 's/^Query response: //')

# 1c. Create the privacy budget for (user, dataset), total epsilon 10.0.
#     Without this, step 2 fails with "Error endorsing txn: ... code = Aborted"; the real
#     reason is only in the peer log: "LogQuery: consumeBudget: readBudget: no budget found".
#     One-time per (user, dataset). On re-runs it fails with "budget already exists" -
#     that is expected and harmless; use PrivacyBudgetContract:UpdateBudget to top up.
curl -b /tmp/fabric-cookies.txt http://localhost:3000/client/invoke -H "Content-Type: application/json" -d '{"channelid": "dt4h", "chaincodeid": "dt4hCC", "function": "PrivacyBudgetContract:InitializeBudget", "args": ["$USER_ID", "test-dataset", "10.0"]}'

# 2. Record a first query on the ledger
curl -b /tmp/fabric-cookies.txt http://localhost:3000/client/invoke -H "Content-Type: application/json" -d '{"channelid": "dt4h", "chaincodeid": "dt4hCC", "function": "LogQuery", "args": ["test-dataset", "select metadata from test-dataset where age>50", "0.5"]}'

# 3. Record a second query on the ledger.
#    The sleep is required: /client/invoke returns without waiting for commit, so a
#    back-to-back invoke re-reads the pre-commit budget and is dropped at validation with
#    MVCC_READ_CONFLICT - while still printing a transaction ID as if it had succeeded.
#
#    KNOWN FLAKE, retry if you hit it: any write may fail with
#      "failed to collect enough transaction endorsements"
#    The chaincode stamps createdAt/updatedAt/timestamp with nowUTC() -> time.Now(), so
#    each endorsing peer writes a different second and the read/write sets do not match
#    the MAJORITY endorsement policy. It only succeeds when all peers land in the same
#    second. Fix belongs in dt4hCC: use ctx.GetStub().GetTxTimestamp() instead.
sleep 3
curl -b /tmp/fabric-cookies.txt http://localhost:3000/client/invoke -H "Content-Type: application/json" -d '{"channelid": "dt4h", "chaincodeid": "dt4hCC", "function": "LogQuery", "args": ["test-dataset", "select metadata from test-dataset where diagnosis=heart-failure", "1.25"]}'
sleep 3

# 4. Read back the history of the two queries above.
#    GetMyHistory uses the caller's identity and takes no args.
curl -b /tmp/fabric-cookies.txt "http://localhost:3000/client/query?channelid=dt4h&chaincodeid=dt4hCC&function=GetMyHistory"

# 4b. Same thing via GetUserHistory - args must be the client ID from step 1b, not the e-mail.
#     Passing the e-mail matches no records and the chaincode then fails marshalling its
#     own response: "return.queries: Invalid type. Expected: array, given: null".
curl -G -b /tmp/fabric-cookies.txt "http://localhost:3000/client/query" --data-urlencode "channelid=dt4h" --data-urlencode "chaincodeid=dt4hCC" --data-urlencode "function=GetUserHistory" --data-urlencode "args=$USER_ID"

# 4c. Remaining privacy budget for the dataset
curl -G -b /tmp/fabric-cookies.txt "http://localhost:3000/client/query" --data-urlencode "channelid=dt4h" --data-urlencode "chaincodeid=dt4hCC" --data-urlencode "function=PrivacyBudgetContract:GetBudget" --data-urlencode "args=$USER_ID" --data-urlencode "args=test-dataset"

# 5. (optional) Close the gateway and drop the session
curl -b /tmp/fabric-cookies.txt http://localhost:3000/client/close
