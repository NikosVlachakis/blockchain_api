# Open a gateway session (enrollmentId + keystore secret for that identity)
curl -c /tmp/fabric-cookies.txt http://localhost:3000/client/ \
  -H "Content-Type: application/json" \
  -d '{"enrollmentId":"blockclient","secret":"blockclientpw"}'

# WhoAmI — returns MSP/enrollmentId (e.g. BscMSP/blockclient)
curl -b /tmp/fabric-cookies.txt \
  "http://localhost:3000/client/query?channelid=dt4h&chaincodeid=dt4hCC&function=PrivacyBudgetContract:WhoAmI"

# LogQuery — 2 args only: datasetID, queryBody (ε comes from SetDatasetQueryCost)
curl -b /tmp/fabric-cookies.txt http://localhost:3000/client/invoke \
  -H "Content-Type: application/json" \
  -d '{"channelid":"dt4h","chaincodeid":"dt4hCC","function":"LogQuery","args":["test-dataset","select metadata from test-dataset"]}'

# Own history (empty users return {"queries":[]})
